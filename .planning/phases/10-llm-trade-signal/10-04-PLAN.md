**Engine semantics (from RESEARCH "PM-to-Ticker Mapping Schema"):**

- Match an incoming `MarketSnapshot` against the mapping table in `config/pm-market-mappings.json`.
- A mapping matches a market if **all non-null `match.*` fields** match:
  - `eventSlug` — exact match on `snapshot.eventSlug` (preferred join key — stable across weekly versions)
  - `slugPrefix` — case-insensitive prefix on `snapshot.slug`
  - `questionContains` — case-insensitive substring on `snapshot.question`
- Multiple mappings may match one market — apply **all** of them. The same ticker appearing in multiple mappings sums.
- `interpretation: "yesPp"` → pass through: `contributedScore = movePp * directionSign * weight`
  - `directionSign = direction === "long" ? +1 : -1`
  - `movePp = snapshot.oneHourPriceChange * 100`
- `interpretation: "noPp"` → invert: `contributedScore = -movePp * directionSign * weight`
  - Rationale: the bullish-for-ticker scenario is "No resolves"; a fall in Yes price (Yes losing) is the bullish signal.
- Before matching, run the market through the EXCLUSION_KEYWORDS regex (sports/entertainment). If it matches an exclusion keyword in `snapshot.question`, **skip the mapping pass entirely** — neither emit a ticker signal nor a proposal.

**Module:**

```typescript
import * as fs from "node:fs/promises";
import * as path from "node:path";
import { EXCLUSION_KEYWORDS } from "../polymarket/relevance-filter.js";
import { JsonlStore } from "../storage/jsonl-store.js";
import type { MarketSnapshot } from "../polymarket/types.js";
import type { PmMapping, PmMappingProposal } from "./types.js";

export interface TickerSignal {
  ticker: string;
  direction: "long" | "short";
  weight: number;
  /** Signed score: movePp × sign(direction) × weight, with interpretation applied. */
  contributedScore: number;
  /** Original 1h price change in pp (signed) — for traceability. */
  movePp: number;
  /** The market that produced this signal. */
  sourceMarketId: string;
  sourceSlug: string;
  sourceEventSlug?: string;
  sourceQuestion: string;
  sourceVolume24hr: number;
  /** Which mapping rule matched (or "llm-proposal" if proposed). */
  matchedBy: string;
}

export interface PmMappingEngineOptions {
  configPath?: string;        // default "./config/pm-market-mappings.json"
  dataDir?: string;           // default "./data/intel" — for proposal persistence
}

interface ConfigFileShape {
  version: number;
  lastUpdated: string;
  mappings: PmMapping[];
  proposed: unknown[];
}

export class PmMappingEngine {
  private readonly configPath: string;
  private readonly proposalStore: JsonlStore<PmMappingProposal>;
  private mappings: PmMapping[] | null = null;

  constructor(options: PmMappingEngineOptions = {}) {
    this.configPath = options.configPath ?? path.resolve("./config/pm-market-mappings.json");
    this.proposalStore = new JsonlStore<PmMappingProposal>(
      options.dataDir ?? "./data/intel",
      "pm-mappings-proposed",
    );
  }

  /**
   * Map a single market. Returns:
   *   - tickerSignals[] when one or more mappings matched (may be empty if matched but tickers cleared)
   *   - { unmatched: true } when NO mapping matched (caller may persist a proposal)
   *   - { excluded: true } when the market's question contains an EXCLUSION_KEYWORDS hit (skip silently)
   */
  async mapMarket(snapshot: MarketSnapshot): Promise<
    | { excluded: true }
    | { unmatched: true }
    | { tickerSignals: TickerSignal[] }
  > {
    if (this.isExcluded(snapshot)) return { excluded: true };
    const mappings = await this.loadMappings();
    const hits = mappings.filter((m) => this.matches(m, snapshot));
    if (hits.length === 0) return { unmatched: true };
    const tickerSignals: TickerSignal[] = [];
    const movePp = (snapshot.oneHourPriceChange ?? 0) * 100;
    for (let i = 0; i < hits.length; i++) {
      const m = hits[i];
      const interpSign = m.interpretation === "noPp" ? -1 : 1;
      for (const t of m.tickers) {
        const dirSign = t.direction === "long" ? 1 : -1;
        const contributedScore = movePp * dirSign * t.weight * interpSign;
        tickerSignals.push({
          ticker: t.ticker,
          direction: t.direction,
          weight: t.weight,
          contributedScore,
          movePp,
          sourceMarketId: snapshot.id,
          sourceSlug: snapshot.slug,
          sourceEventSlug: snapshot.eventSlug,
          sourceQuestion: snapshot.question,
          sourceVolume24hr: snapshot.volume24hr,
          matchedBy: this.matchKey(m),
        });
      }
    }
    return { tickerSignals };
  }

  /**
   * Convenience: map many snapshots; collect signals, persist proposals.
   * Caller passes a single timestamp for the batch — used in proposal records.
   */
  async mapMarkets(snapshots: MarketSnapshot[]): Promise<{
    tickerSignals: TickerSignal[];
    proposals: PmMappingProposal[];
    excludedCount: number;
  }> {
    const tickerSignals: TickerSignal[] = [];
    const proposals: PmMappingProposal[] = [];
    let excludedCount = 0;
    for (const s of snapshots) {
      const r = await this.mapMarket(s);
      if ("excluded" in r) { excludedCount++; continue; }
      if ("unmatched" in r) {
        proposals.push({
          marketId: s.id,
          slug: s.slug,
          eventSlug: s.eventSlug,
          question: s.question,
          proposedTickers: [],          // populated later by the article scorer's LLM fallback
          interpretationSuggestion: "yesPp",
          proposedAt: new Date().toISOString(),
        });
        continue;
      }
      tickerSignals.push(...r.tickerSignals);
    }
    if (proposals.length > 0) {
      await this.proposalStore.appendMany(proposals);
    }
    return { tickerSignals, proposals, excludedCount };
  }

  private async loadMappings(): Promise<PmMapping[]> {
    if (this.mappings !== null) return this.mappings;
    try {
      const raw = await fs.readFile(this.configPath, "utf8");
      const parsed = JSON.parse(raw) as ConfigFileShape;
      this.mappings = parsed.mappings ?? [];
      return this.mappings;
    } catch (err) {
      if ((err as NodeJS.ErrnoException).code === "ENOENT") {
        console.warn(`[PmMappingEngine] config file not found at ${this.configPath} — mapping disabled`);
        this.mappings = [];
        return [];
      }
      throw err;
    }
  }

  private matches(m: PmMapping, s: MarketSnapshot): boolean {
    if (m.match.eventSlug !== null && s.eventSlug !== m.match.eventSlug) return false;
    if (m.match.slugPrefix !== null && !s.slug.toLowerCase().startsWith(m.match.slugPrefix.toLowerCase())) return false;
    if (m.match.questionContains !== null && !s.question.toLowerCase().includes(m.match.questionContains.toLowerCase())) return false;
    // If all three are null, the rule matches NOTHING (refuse to fire a "catch-all" mapping).
    if (m.match.eventSlug === null && m.match.slugPrefix === null && m.match.questionContains === null) return false;
    return true;
  }

  private isExcluded(s: MarketSnapshot): boolean {
    const haystack = s.question.toLowerCase();
    return EXCLUSION_KEYWORDS.some((k) => haystack.includes(k.toLowerCase()));
  }

  private matchKey(m: PmMapping): string {
    const parts: string[] = [];
    if (m.match.eventSlug) parts.push(`eventSlug=${m.match.eventSlug}`);
    if (m.match.slugPrefix) parts.push(`slugPrefix=${m.match.slugPrefix}`);
    if (m.match.questionContains) parts.push(`questionContains="${m.match.questionContains}"`);
    return parts.join("&") || "(empty)";
  }

  /** Invalidate the in-memory mapping cache (used after operator runs `intel pm-mappings review`). */
  invalidateCache(): void {
    this.mappings = null;
  }
}
```

6. **All-null match rule never fires**: stage a mapping with `match: { eventSlug: null, slugPrefix: null, questionContains: null }`. Verify it doesn't match any snapshot.
7. **Multiple mappings on one market**: stage two mappings that both match (e.g., one on eventSlug, one on slugPrefix); verify both apply (signals from both rules appear).
