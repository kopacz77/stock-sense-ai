```typescript
import * as fs from "node:fs/promises";
import * as path from "node:path";
import { JsonlStore } from "../storage/jsonl-store.js";
import type {
  CatalystFlag,
  DigestPayload,
  ScoredArticle,
} from "../signal/types.js";
import type { MarketSnapshot } from "../polymarket/types.js";

export interface DigestBuilderOptions {
  dataDir?: string;     // default "./data/intel"
}

export class DigestBuilder {
  private readonly dataDir: string;
  private readonly scoredStore: JsonlStore<ScoredArticle>;
  private readonly snapshotStore: JsonlStore<MarketSnapshot>;

  constructor(options: DigestBuilderOptions = {}) {
    this.dataDir = options.dataDir ?? "./data/intel";
    this.scoredStore = new JsonlStore<ScoredArticle>(this.dataDir, "scored-articles");
    this.snapshotStore = new JsonlStore<MarketSnapshot>(this.dataDir, "polymarket-snapshots");
  }

  /**
   * Build a digest for the given flavor.
   *
   * MORNING window  : prior day 16:00 ET → today 08:30 ET (covers overnight + pre-market)
   * MIDDAY window   : today 08:30 ET → 12:30 ET
   * CLOSE window    : today 12:30 ET → 15:30 ET
   *
   * `topStories`: top 1-2 (article × ticker) scored rows by materiality DESC, secondary sort |sentiment| DESC.
   * `upcomingCalendar`: catalyst-flags non-archived with expectedDate within next 24h.
   * `pmMovers`: MIDDAY + CLOSE only. Top 2 PM markets by |oneHourPriceChange| × log10(volume24hr+1).
   */
  async build(flavor: "MORNING" | "MIDDAY" | "CLOSE", now: Date = new Date()): Promise<DigestPayload> {
    // 1) Score-articles window
    const windowStart = computeWindowStart(flavor, now);
    const scored = await this.loadScoredSince(windowStart, now);
    const topStories = await this.pickTopStories(scored, 2);

    // 2) Upcoming calendar (24h)
    const upcomingCalendar = await this.loadUpcomingCalendar(now, 24);

    // 3) PM movers (MIDDAY/CLOSE)
    const pmMovers = flavor === "MORNING" ? undefined : await this.loadPmMovers(2);

    return {
      flavor,
      builtAt: now.toISOString(),
      topStories,
      upcomingCalendar,
      pmMovers,
    };
  }

  private async loadScoredSince(since: Date, until: Date): Promise<ScoredArticle[]> {
    // Read today + yesterday's scored files; filter by scoredAt in [since, until].
    const today = await this.scoredStore.readDay(until);
    const yest = await this.scoredStore.readDay(new Date(until.getTime() - 24 * 60 * 60 * 1000));
    const ms = (s: ScoredArticle) => Date.parse(s.scoredAt);
    return [...yest, ...today].filter((s) => ms(s) >= since.getTime() && ms(s) <= until.getTime());
  }

  private async pickTopStories(
    scored: ScoredArticle[],
    n: number,
  ): Promise<DigestPayload["topStories"]> {
    // Dedup by sourceArticleId (one row per article in the digest, not per ticker)
    const byArticle = new Map<string, ScoredArticle>();
    for (const s of scored) {
      const existing = byArticle.get(s.sourceArticleId);
      // Take the row with highest materiality for the article
      if (!existing || s.materiality > existing.materiality) byArticle.set(s.sourceArticleId, s);
    }
    const ranked = Array.from(byArticle.values()).sort((a, b) => {
      if (b.materiality !== a.materiality) return b.materiality - a.materiality;
      return Math.abs(b.sentiment) - Math.abs(a.sentiment);
    });
    const top = ranked.slice(0, n);
    // We need headline/publisher/url/tickers — these live in the original NewsArticle.
    // Approach: read today's news-*.jsonl and join by id.
    const articleMeta = await this.loadArticleMeta(top.map((s) => s.sourceArticleId));
    return top.map((s) => {
      const m = articleMeta.get(s.sourceArticleId);
      return {
        articleId: s.sourceArticleId,
        headline: m?.headline ?? "(headline missing)",
        publisher: m?.publisher ?? m?.source ?? "(unknown)",
        tickers: m?.tickers ?? (s.ticker ? [s.ticker] : []),
        sentiment: s.sentiment,
        materiality: s.materiality,
        rationale: this.buildRationale(s),
        url: m?.url ?? "",
      };
    });
  }

  private async loadArticleMeta(ids: string[]): Promise<Map<string, {
    headline: string; publisher?: string; source: string; tickers: string[]; url: string;
  }>> {
    if (ids.length === 0) return new Map();
    const idSet = new Set(ids);
    const newsStore = new JsonlStore<{
      id: string; headline: string; publisher?: string; source: string; tickers: string[]; url: string;
    }>(this.dataDir, "news");
    const today = await newsStore.readDay(new Date());
    const yest = await newsStore.readDay(new Date(Date.now() - 24 * 60 * 60 * 1000));
    const map = new Map<string, {
      headline: string; publisher?: string; source: string; tickers: string[]; url: string;
    }>();
    for (const a of [...yest, ...today]) {
      if (idSet.has(a.id) && !map.has(a.id)) map.set(a.id, { headline: a.headline, publisher: a.publisher, source: a.source, tickers: a.tickers, url: a.url });
    }
    return map;
  }

  private buildRationale(s: ScoredArticle): string {
    const dir = s.sentiment >= 0 ? "bullish" : "bearish";
    const mag = s.materiality >= 0.7 ? "high-impact" : s.materiality >= 0.4 ? "notable" : "context";
    const themesPart = s.themes.length > 0 ? ` themes: ${s.themes.slice(0, 2).join(", ")}.` : "";
    return `${mag} ${dir} (sent=${s.sentiment.toFixed(2)}, mat=${s.materiality.toFixed(2)}).${themesPart}`;
  }

  private async loadUpcomingCalendar(now: Date, hours: number): Promise<DigestPayload["upcomingCalendar"]> {
    const files = await fs.readdir(this.dataDir).catch(() => [] as string[]);
    const matching = files.filter((f) => /^catalyst-flags-\d{4}-\d{2}-\d{2}\.jsonl$/.test(f));
    const all: CatalystFlag[] = [];
    for (const f of matching) {
      const content = await fs.readFile(path.join(this.dataDir, f), "utf8").catch(() => "");
      for (const line of content.split("\n")) {
        const t = line.trim();
        if (!t) continue;
        try { all.push(JSON.parse(t) as CatalystFlag); } catch { /* skip */ }
      }
    }
    // Dedup by id — latest wins
    const byId = new Map<string, CatalystFlag>();
    for (const c of all) {
      const existing = byId.get(c.id);
      const cTs = Date.parse(c.lastRefinedAt ?? c.firstSeenAt);
      const eTs = existing ? Date.parse(existing.lastRefinedAt ?? existing.firstSeenAt) : -Infinity;
      if (!existing || cTs > eTs) byId.set(c.id, c);
    }
    const cutoff = now.getTime() + hours * 60 * 60 * 1000;
    return Array.from(byId.values())
      .filter((c) => c.archived !== true)
      .filter((c) => Date.parse(c.expectedDate) >= now.getTime() && Date.parse(c.expectedDate) <= cutoff)
      .sort((a, b) => b.magnitudePrior - a.magnitudePrior)
      .slice(0, 5)
      .map((c) => ({
        eventId: c.id,
        type: c.type,
        label: humanLabel(c),
        expectedDate: c.expectedDate,
        expectedTimeEt: c.expectedTimeEt,
        magnitudePrior: c.magnitudePrior,
        affectedTickers: [...(c.tickers ?? []), ...(c.affectedSectors ?? [])],
      }));
  }

  private async loadPmMovers(n: number): Promise<DigestPayload["pmMovers"]> {
    const today = await this.snapshotStore.readDay(new Date());
    // Take the latest snapshot per marketId, then rank by |oneHourPriceChange| × log10(volume+1)
    const byId = new Map<string, MarketSnapshot>();
    for (const s of today) {
      const cur = byId.get(s.id);
      // Snapshots arrive in fetchedAt order in the file; latest write wins.
      if (!cur) byId.set(s.id, s);
      else byId.set(s.id, s);
    }
    const ranked = Array.from(byId.values()).sort((a, b) => {
      const aa = Math.abs(a.oneHourPriceChange ?? 0) * Math.log10(Math.max(a.volume24hr, 1));
      const bb = Math.abs(b.oneHourPriceChange ?? 0) * Math.log10(Math.max(b.volume24hr, 1));
      return bb - aa;
    });
    return ranked.slice(0, n).map((m) => ({
      marketId: m.id,
      question: m.question,
      movePp: (m.oneHourPriceChange ?? 0) * 100,
      volume24hr: m.volume24hr,
    }));
  }
}

function computeWindowStart(flavor: "MORNING" | "MIDDAY" | "CLOSE", now: Date): Date {
  // Approximate boundaries in absolute time. ET-time-of-day check is in the scheduler.
  const day = 24 * 60 * 60 * 1000;
  if (flavor === "MORNING") return new Date(now.getTime() - 16 * 60 * 60 * 1000); // overnight window
  if (flavor === "MIDDAY") return new Date(now.getTime() - 4 * 60 * 60 * 1000);   // ~since open
  return new Date(now.getTime() - 3 * 60 * 60 * 1000);                            // since midday
}

function humanLabel(c: CatalystFlag): string {
  const tickerPart = (c.tickers ?? []).length > 0 ? ` (${(c.tickers ?? []).join(",")})` : "";
  return `${c.type.replace(/_/g, " ").toUpperCase()}${tickerPart}`;
}

/**
 * Render a DigestPayload to Markdown (Telegram MarkdownV1).
 */
export function renderDigestMarkdown(p: DigestPayload): string {
  const lines: string[] = [];
  const title =
    p.flavor === "MORNING" ? "*Morning Brief*" :
    p.flavor === "MIDDAY" ? "*Mid-Day Update*" :
    "*Pre-Close Recap*";
  lines.push(title, "");

  if (p.topStories.length === 0) {
    lines.push("_No high-materiality stories this window._");
  } else {
    lines.push("*Top stories*");
    for (const s of p.topStories) {
      const tickerStr = s.tickers.length > 0 ? `[${s.tickers.join(", ")}]` : "[—]";
      lines.push(`• ${tickerStr} _${escapeMd(s.headline)}_`);
      lines.push(`  ${s.rationale}  ([read](${s.url}))`);
    }
  }
  lines.push("");
  lines.push("*Next 24h calendar*");
  if (p.upcomingCalendar.length === 0) {
    lines.push("_(nothing scheduled)_");
  } else {
    for (const e of p.upcomingCalendar) {
      const when = e.expectedTimeEt ? `${e.expectedDate} ${e.expectedTimeEt} ET` : e.expectedDate;
      lines.push(`• \`${e.type}\` ${e.label} — ${when} (m=${e.magnitudePrior})`);
    }
  }
  if (p.pmMovers && p.pmMovers.length > 0) {
    lines.push("");
    lines.push("*PM movers*");
    for (const m of p.pmMovers) {
      const sign = m.movePp >= 0 ? "+" : "";
      lines.push(`• ${escapeMd(m.question)} — ${sign}${m.movePp.toFixed(1)} pp, $${compact(m.volume24hr)}`);
    }
  }
  return lines.join("\n");
}

function escapeMd(s: string): string { return s.replace(/([_*[\]`])/g, "\\$1"); }
function compact(n: number): string {
  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`;
  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}k`;
  return n.toFixed(0);
}
```

```typescript
private async checkDigests(): Promise<void> {
  const now = new Date();
  const etHM = etHourMinute(now);   // "08:31"
  const flavor = matchDigestFlavor(etHM, this.digestTimes); // "MORNING" | "MIDDAY" | "CLOSE" | null
  if (!flavor) return;

  // Build + send via alerter. Alerter is idempotent (already-fired returns false silently).
  const builder = new DigestBuilder({ dataDir: this.options.dataDir ?? "./data/intel" });
  const payload = await builder.build(flavor, now);
  const alerter = new IntelligenceAlerter(this.options.dataDir ?? "./data/intel");
  const sent = await alerter.sendDigest(payload);
  if (sent) console.log(`[intel-scheduler] ${flavor} digest sent at ET ${etHM}`);
}
```

Where `matchDigestFlavor(etHM, digestTimes)` returns the flavor if `etHM` is within ±1 minute of one of the targets, else null.

Where `etHourMinute(d: Date): string` returns "HH:MM" in America/New_York (use `Intl.DateTimeFormat`).

Call `checkDigests()` from the heartbeat AFTER the cycle gate (independent code paths):
```typescript
private async heartbeat(onCycle?: (result: CycleResult) => void): Promise<void> {
  if (!this.running) return;
  // Existing cycle-gate logic...
  // After cycle handling (whether or not a cycle ran), check digests:
  try { await this.checkDigests(); } catch (err) {
    console.error("[intel-scheduler] digest check failed:", err instanceof Error ? err.message : err);
  }
}
```

This way digests can fire even on ticks where no cycle ran (the cadence gates are independent).

```typescript
function evaluateBreakGlass(
  alerts: IntelligenceAlert[],
  scored: ScoredArticle[],
  catalysts: CatalystFlag[],
  now: Date,
): { trigger: IntelligenceAlert | DigestPayload | null; reason: string } {
```
