  /** Sequential gate — only one scoreArticle() may be in flight at a time. */

  constructor(options: ArticleScorerOptions) {
    this.client = new OpenAI({
      apiKey: options.apiKey || "lm-studio",
      baseURL: options.endpoint,
      timeout: options.timeoutMs ?? 60_000,
      maxRetries: 0,
    });
    this.model = options.model;
    this.maxOutputTokens = options.maxOutputTokens ?? 1024;
    this.temperature = options.temperature ?? 0.2;

    if (options.concurrency !== undefined && options.concurrency > 1) {
      console.warn(
        `[ArticleScorer] concurrency=${options.concurrency} requested but clamped to 1 — ` +
          "local Qwen on single GPU saturates with parallel calls.",
      );
    }
  }

   * Score a single article. Fans out one record per (article × ticker).
   * Sequential: chains through `this.inFlight` so callers can issue back-to-back
   * `await scoreArticle(...)` without worrying about overlap.
   *
   * On LLM failure: throws. Caller enqueues to backlog.
   */
  async scoreArticle(
    article: NewsArticle,
    context: ScoringContext,
  ): Promise<ScoredArticle[]> {
    // Sequential gate.
    const prev = this.inFlight;
    let release!: () => void;
    this.inFlight = new Promise<void>((r) => { release = r; });
    try {
      await prev;
      return await this.scoreArticleImpl(article, context);
    } finally {
      release();
    }
  }

  private async scoreArticleImpl(
    article: NewsArticle,
    context: ScoringContext,
  ): Promise<ScoredArticle[]> {
    const userMessage = this.buildUserMessage(article, context);

    // Note: response_format omitted — see LlmCorrelator line 117-119 for the
    // LM Studio JSON-mode quirk that this works around.
    const response = await this.client.chat.completions.create({
      model: this.model,
      max_tokens: this.maxOutputTokens,
      temperature: this.temperature,
      messages: [
        { role: "system", content: SYSTEM_PROMPT_V1 },
        { role: "user", content: userMessage },
      ],
    });

    const text = response.choices[0]?.message?.content ?? "";
    const parsed = parseScorerResponse(text);

    return fanOutScoredArticle(article, parsed, context, this.model);
  }

The backlog absorbs LLM-down windows. When `ArticleScorer.scoreArticle` throws, the cycle-runner (Plan 06) calls `backlog.enqueue(article, pmContext, errorMessage)`. When the LLM recovers, the next cycle calls `backlog.drain(scorer, context, maxN=50)` AFTER scoring the fresh batch (so fresh signal isn't starved).

**File: single-file rolling backlog at `data/intel/score-backlog.jsonl`.** Not date-rotated (it's ephemeral — entries leave when scored). Re-writes the whole file on every drain (atomic via temp-file rename).

```typescript
import * as fs from "node:fs/promises";
import * as path from "node:path";
import type { NewsArticle } from "../news/types.js";
import type { MarketSnapshot } from "../polymarket/types.js";
import type { ScoreBacklogEntry, ScoredArticle } from "./types.js";
import type { ArticleScorer, ScoringContext } from "./article-scorer.js";

export interface ScoreBacklogOptions {
  /** Directory containing the backlog file. Default ./data/intel */
  dataDir?: string;
  /** Filename. Default "score-backlog.jsonl" */
  filename?: string;
}

export interface DrainResult {
  scored: number;
  failed: number;
  remaining: number;
  oldestAgeMs: number | null;
  scoredRecords: ScoredArticle[];
}

export class ScoreBacklog {
  private readonly file: string;
  private cache: ScoreBacklogEntry[] | null = null;

  constructor(options: ScoreBacklogOptions = {}) {
    const dir = options.dataDir ?? "./data/intel";
    const name = options.filename ?? "score-backlog.jsonl";
    this.file = path.resolve(path.join(dir, name));
  }

  async enqueue(article: NewsArticle, pmContext: MarketSnapshot[], errorMessage?: string): Promise<void> {
    const entries = await this.load();
    // De-dup by article.id — if already present, bump attempts + update lastErrorMessage but keep enqueuedAt.
    const existing = entries.find((e) => e.article.id === article.id);
    if (existing) {
      existing.attempts += 1;
      if (errorMessage) existing.lastErrorMessage = errorMessage;
    } else {
      entries.push({
        enqueuedAt: new Date().toISOString(),
        attempts: 1,
        lastErrorMessage: errorMessage,
        article,
        pmContext,
      });
    }
    await this.writeAll(entries);
  }

  /** Process up to maxN entries oldest-first. Successful score removes entry. */
  async drain(
    scorer: ArticleScorer,
    context: ScoringContext,
    maxN = 50,
  ): Promise<DrainResult> {
    const entries = await this.load();
    entries.sort((a, b) => Date.parse(a.enqueuedAt) - Date.parse(b.enqueuedAt));
    const slice = entries.slice(0, maxN);
    const remaining: ScoreBacklogEntry[] = entries.slice(maxN);
    const scoredRecords: ScoredArticle[] = [];
    let scored = 0;
    let failed = 0;

    for (const entry of slice) {
      try {
        const ctx: ScoringContext = { ...context, pmContext: entry.pmContext };
        const records = await scorer.scoreArticle(entry.article, ctx);
        scoredRecords.push(...records);
        scored += 1;
      } catch (err) {
        failed += 1;
        entry.attempts += 1;
        entry.lastErrorMessage = err instanceof Error ? err.message : String(err);
        remaining.push(entry);
        // Stop early on first failure of a drain cycle — LLM likely still down.
        // Add remaining unprocessed `slice` entries back to remaining.
        const idx = slice.indexOf(entry);
        remaining.push(...slice.slice(idx + 1));
        break;
      }
    }

    await this.writeAll(remaining);
    const oldestAgeMs = remaining.length === 0 ? null : Date.now() - Date.parse(
      remaining.reduce((acc, e) => Date.parse(e.enqueuedAt) < Date.parse(acc.enqueuedAt) ? e : acc).enqueuedAt
    );
    return { scored, failed, remaining: remaining.length, oldestAgeMs, scoredRecords };
  }

  async size(): Promise<number> {
    const entries = await this.load();
    return entries.length;
  }

  async oldestAgeMs(): Promise<number | null> {
    const entries = await this.load();
    if (entries.length === 0) return null;
    const oldest = entries.reduce(
      (acc, e) => Date.parse(e.enqueuedAt) < Date.parse(acc.enqueuedAt) ? e : acc,
    );
    return Date.now() - Date.parse(oldest.enqueuedAt);
  }

  /** Read all entries from disk. Returns [] if file doesn't exist. */
  private async load(): Promise<ScoreBacklogEntry[]> {
    if (this.cache !== null) return [...this.cache];
    try {
      const raw = await fs.readFile(this.file, "utf8");
      const lines = raw.split("\n").filter((l) => l.trim().length > 0);
      this.cache = lines.map((l) => JSON.parse(l) as ScoreBacklogEntry);
      return [...this.cache];
    } catch (err) {
      if ((err as NodeJS.ErrnoException).code === "ENOENT") {
        this.cache = [];
        return [];
      }
      throw err;
    }
  }

  /** Atomic write: temp file + rename. */
  private async writeAll(entries: ScoreBacklogEntry[]): Promise<void> {
    await fs.mkdir(path.dirname(this.file), { recursive: true });
    const tmp = `${this.file}.tmp`;
    const body = entries.map((e) => JSON.stringify(e)).join("\n") + (entries.length > 0 ? "\n" : "");
    await fs.writeFile(tmp, body, "utf8");
    await fs.rename(tmp, this.file);
    this.cache = entries;
  }
}
```

**Unit tests** in `src/market-intelligence/signal/__tests__/score-backlog.test.ts`:

Use a temp dir per test (`fs.mkdtemp(path.join(os.tmpdir(), "score-backlog-"))`) so tests don't collide.

Use a tiny fake scorer that returns a canned ScoredArticle[] on each call. Required cases:
1. **enqueue → load round-trip**: enqueue one entry, instantiate a fresh ScoreBacklog (no cache), load, see the entry.
2. **enqueue dedup**: enqueue same article id twice, size() returns 1, attempts bumped to 2.
3. **drain oldest-first**: enqueue 3 entries with staggered timestamps; drain with maxN=2; assert the 2 oldest were scored, 1 remains.
4. **drain cap**: enqueue 10 entries; drain maxN=3; assert 3 scored, 7 remain.
5. **drain stops on first failure**: scorer throws on the 2nd call; assert 1 scored, 9 remain (10 enqueued).
6. **oldestAgeMs**: enqueue 2 entries 1s apart; assert oldestAgeMs ≥ 1000.
7. **empty drain**: drain on empty backlog returns scored=0, remaining=0, oldestAgeMs=null.

Coding notes:
- `JsonlStore` is not used here — single-file, occasional rewrites, not append-only.
- Cache invalidation is naive (only `load` populates it). Fine for single-process scheduler.
- Atomic write via temp-file rename prevents a half-written backlog if the process is killed mid-drain.
